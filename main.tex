% Compile with pdfLaTeX in Overleaf. Change this one setting:
% outline = working outline; manuscript = IEEE journal draft; both = outline + draft.
\def\paperview{both}

\newif\ifshowoutline
\newif\ifshowmanuscript
\def\outlineview{outline}
\def\manuscriptview{manuscript}
\def\bothview{both}
\ifx\paperview\outlineview
  \showoutlinetrue
\else\ifx\paperview\manuscriptview
  \showmanuscripttrue
\else\ifx\paperview\bothview
  \showoutlinetrue
  \showmanuscripttrue
\else
  \errmessage{Invalid paperview: use outline, manuscript, or both}
\fi\fi\fi

\ifshowmanuscript
  \documentclass[10pt,journal]{IEEEtran}
\else
  \documentclass[10pt]{article}
\fi
\input{paper/preamble}

\begin{document}

\ifshowoutline
  \onecolumn
  \begingroup
  \renewcommand{\thesection}{\arabic{section}}
  \renewcommand{\thesubsection}{\thesection.\arabic{subsection}}
  \renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}
  % Distinct PDF destinations when section numbering restarts in the manuscript.
  \renewcommand{\theHsection}{outline.\arabic{section}}
  \section*{Working outline}
  \input{paper/outline}
  \endgroup
\fi

\ifshowmanuscript
  \clearpage
  \setcounter{section}{0}
  \setcounter{equation}{0}
  \setcounter{figure}{0}
  \setcounter{table}{0}
  \setcounter{footnote}{0}
  \renewcommand{\theHsection}{manuscript.\arabic{section}}
  \twocolumn
  \input{paper/manuscript}
\fi

\bibliographystyle{IEEEtran}
\bibliography{paper/references}

\end{document}
